# useList hooks and components

A set of hooks and building blocks for creating list / tree list UIs.

The basic idea is that hooks take all the complex logic on themselves, and all you have to do is implement the "dumb" components of the view.

[`TreeList`](/docs/lab-treelist--docs) and [`TreeSelect`](/docs/lab-treeselect--docs) are ready-made compositions built on these primitives — useful as references and as drop-in UI when they fit your needs.

## Quick code snippets

### Flatten items:

```tsx
import * as React from 'react';
import {
  type ListItemType,
  ListContainerView,
  ListItemView,
  getItemRenderState,
  getListItemClickHandler,
  useList,
  useListKeydown,
} from '@gravity-ui/tree-select';

const items: ListItemType<string>[] = ['one', 'two', 'free', 'four', 'five'];

function List() {
  const containerRef = React.useRef<HTMLDivElement>(null);

  const list = useList({items});
  const onItemClick = getListItemClickHandler({list, multiple: true});
  useListKeydown({onItemClick, containerRef, list});

  return (
    <ListContainerView ref={containerRef}>
      {list.structure.visibleFlattenIds.map((id) => {
        const {props} = getItemRenderState({
          id,
          mapItemDataToContentProps: (title) => ({title}),
          onItemClick,
          list,
        });

        return <ListItemView key={id} {...props} />;
      })}
    </ListContainerView>
  );
}
```

### Tree item structure:

```tsx
import * as React from 'react';
import {
  type ListItemType,
  ListContainerView,
  ListItemRecursiveRenderer,
  ListItemView,
  getItemRenderState,
  getListItemClickHandler,
  useList,
} from '@gravity-ui/tree-select';

const items: ListItemType<string>[] = [
  {data: 'one'},
  {data: 'two', children: [{data: 'tree', children: [{data: 'four'}, {data: 'five'}]}]},
];

function List() {
  const containerRef = React.useRef<HTMLDivElement>(null);
  const list = useList({items});
  const onItemClick = getListItemClickHandler({list});

  return (
    <ListContainerView ref={containerRef}>
      {list.structure.items.map((itemSchema, index) => (
        <ListItemRecursiveRenderer
          key={index}
          itemSchema={itemSchema}
          id={list.structure.rootIds[index]}
          list={list}
        >
          {(id) => {
            const {props} = getItemRenderState({
              id,
              mapItemDataToContentProps: (title) => ({title}),
              onItemClick,
              list,
            });

            return <ListItemView {...props} />;
          }}
        </ListItemRecursiveRenderer>
      ))}
    </ListContainerView>
  );
}
```
